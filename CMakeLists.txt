cmake_minimum_required(VERSION 3.24)
project(NPCBehaviorBridge VERSION 0.6.7 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NBB_BUILD_PLUGIN "Build the SKSE plugin (explicit Steam/GOG runtime list)" ON)
option(NBB_BUILD_TESTS "Build local, untracked verification targets" OFF)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/zlib EXCLUDE_FROM_ALL)
add_library(nbb_core STATIC src/core.cpp)
target_include_directories(nbb_core PUBLIC include vendor/json/single_include PRIVATE vendor/zlib "${CMAKE_BINARY_DIR}/vendor/zlib")
target_link_libraries(nbb_core PUBLIC zlibstatic bcrypt)
target_compile_definitions(nbb_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(nbb_core PRIVATE /W4 /permissive- /utf-8)
add_executable(nbb_cli src/cli.cpp)
target_link_libraries(nbb_cli PRIVATE nbb_core)
if(NBB_BUILD_TESTS)
  enable_testing()
  add_executable(nbb_tests tests/core_tests.cpp)
  target_link_libraries(nbb_tests PRIVATE nbb_core)
  target_include_directories(nbb_tests PRIVATE vendor/zlib "${CMAKE_BINARY_DIR}/vendor/zlib")
  add_test(NAME CoreContracts COMMAND nbb_tests)
  add_executable(nbb_inheritance_tests tests/inheritance_tests.cpp)
  target_link_libraries(nbb_inheritance_tests PRIVATE nbb_core)
  add_test(NAME InheritanceContracts COMMAND nbb_inheritance_tests)
  add_executable(nbb_script_tests tests/script_binding_tests.cpp)
  target_link_libraries(nbb_script_tests PRIVATE nbb_core)
  add_test(NAME ScriptBindingContracts COMMAND nbb_script_tests)
endif()
if(NBB_BUILD_PLUGIN)
  set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
  set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
  add_subdirectory(vendor/spdlog EXCLUDE_FROM_ALL)
  set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(BUILD_XAUDIO_WIN8 OFF CACHE BOOL "" FORCE)
  set(BUILD_XAUDIO_WIN10 OFF CACHE BOOL "" FORCE)
  set(BUILD_XAUDIO_REDIST OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/DirectXTK EXCLUDE_FROM_ALL)
  add_library(Microsoft::DirectXTK ALIAS DirectXTK)
  set(spdlog_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake/dependencies" CACHE PATH "" FORCE)
  set(directxtk_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake/dependencies" CACHE PATH "" FORCE)
  set(RAPIDCSV_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/rapidcsv/src" CACHE PATH "" FORCE)
  set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
  set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/CommonLibSSE-NG EXCLUDE_FROM_ALL)
  target_compile_options(CommonLibSSE PRIVATE /MP4)
  add_library(NPCBehaviorBridge SHARED src/plugin.cpp src/script_binding.cpp)
  target_link_libraries(NPCBehaviorBridge PRIVATE nbb_core CommonLibSSE)
  target_compile_definitions(NPCBehaviorBridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(NPCBehaviorBridge PRIVATE /W4 /permissive- /utf-8 /Zi)
  target_link_options(NPCBehaviorBridge PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF)
  target_precompile_headers(NPCBehaviorBridge PRIVATE <RE/Skyrim.h> <SKSE/SKSE.h>)
  if(NBB_BUILD_TESTS)
    set(NBB_ADDRESS_LIBRARY_DIR "" CACHE PATH "Installed address libraries for offline runtime checks")
    add_executable(nbb_runtime_tests tests/runtime_support_tests.cpp)
    target_compile_definitions(nbb_runtime_tests PRIVATE ENABLE_COMMONLIBSSE_TESTING NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(nbb_runtime_tests PRIVATE nbb_core CommonLibSSE)
    target_precompile_headers(nbb_runtime_tests PRIVATE <RE/Skyrim.h> <SKSE/SKSE.h>)
    add_test(NAME RuntimeContracts COMMAND nbb_runtime_tests "${NBB_ADDRESS_LIBRARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/runtime-addresses.json")
  endif()
endif()
